\[
H_{t,4}(z)=\int_0^\infty e^{tu^4}\Phi(u)\cos(zu)\,du,

\Phi(u)=\sum_{n\ge1}(2\pi^2 n^4e^{9u}-3\pi n^2e^{5u})e^{-\pi n^2e^{4u}}.
